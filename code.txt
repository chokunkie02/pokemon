        let state = {
            numDecks: 3, // แก้ไขจำนวนขอนไพ่เหลือ 3 สำรับ (รวม 156 ใบ)

            savedPhase: null,
            isDropMode: false,

            {val: '8', type: 'btn-num'}, {val: '9', type: 'btn-num'}, {val: '10', type: 'btn-face'}, 
            {val: 'J', type: 'btn-face'}, {val: 'Q', type: 'btn-face'}, {val: 'K', type: 'btn-face'},

        function initDeck() {
            const baseDeck = [2, 3, 4, 5, 6, 7, 8, 9, 10, 10, 10, 10, 11];
            state.mainDeck = [];
            for(let i=0; i < 4 * state.numDecks; i++) {
                state.mainDeck = state.mainDeck.concat(baseDeck);
            }
            state.roundDeck = [...state.mainDeck];
            state.discardPile = [];
            historyStack = [];
        }

        function getScore(hand) {
            let score = hand.reduce((a, b) => a + b, 0);
            let aces = hand.filter(c => c === 11).length;
            while (score > 21 && aces > 0) {
                score -= 10;
                aces -= 1;
            }
            return score;
        }

        function parseCard(c) {
            if (c === 'A') return 11;
            if (['J', 'Q', 'K'].includes(c)) return 10;
            return parseInt(c);
        }

        function formatCardValue(val) {
            if (val === 11) return 'A';
            return val.toString();
        }

        // ==========================================
        // 3. AI Simulation
        // ==========================================
        function simulateAction(action, numSims = 3000) { 
            if (state.dealerHand.length === 0) return 0.0;
            let wins = 0;
            let pushes = 0;
            const userHand = state.playerHand;

            for (let i = 0; i < numSims; i++) {
                let simDeck = [...state.roundDeck];
                for (let k = 0; k < Math.min(10, simDeck.length); k++) {
                    const randIdx = k + Math.floor(Math.random() * (simDeck.length - k));
                    [simDeck[k], simDeck[randIdx]] = [simDeck[randIdx], simDeck[k]];
                }
                let drawnCards = simDeck.slice(0, 10);
                
                let pHand = [...userHand];
                let dHand = [...state.dealerHand];

                if (action === 'HIT') {
                    pHand.push(drawnCards.shift());
                    while (getScore(pHand) < 15 && drawnCards.length > 0) pHand.push(drawnCards.shift());
                } else if (action === 'DOUBLE') {
                    pHand.push(drawnCards.shift());
                } else if (action === 'SPLIT') {
                    pHand = [pHand[0], drawnCards.shift()];
                    while (getScore(pHand) < 15 && drawnCards.length > 0) pHand.push(drawnCards.shift());
                }

                let pScore = getScore(pHand);
                if (pScore > 21) continue; 

                let dScore = getScore(dHand);
                while (dScore < 17 && drawnCards.length > 0) {
                    dHand.push(drawnCards.shift());
                    dScore = getScore(dHand);
                }

                if (dScore > 21 || pScore > dScore) wins++;
                else if (dScore === pScore) pushes++;
            }
            return ((wins + (pushes * 0.5)) / numSims) * 100; 
        }

        let aiTimeout = null;
        function runAIAnalysis() {
            if (state.isDropMode) return; 

            document.getElementById('ai-result').innerHTML = "<i>Computing AI odds...</i>";
            setButtonsState(false);
            
            if(aiTimeout) clearTimeout(aiTimeout);
            aiTimeout = setTimeout(() => {
                try {
                    const userHand = state.playerHand;
                    let actions = ['STAND', 'HIT'];
                    if (userHand.length === 2) actions.push('DOUBLE');
                    if (userHand.length === 2 && userHand[0] === userHand[1]) actions.push('SPLIT');

                    let results = [];
                    for (let act of actions) {
                        results.push({ action: act, winRate: simulateAction(act) });
                    }
                    
                    results.sort((a, b) => b.winRate - a.winRate);
                    let bestMove = results[0].action;

                    let html = `Optimal Move: <span class="highlight">${bestMove}</span><br>`;
                    results.forEach(r => {
                        let star = r.action === bestMove ? "⭐" : "&nbsp;&nbsp;";
                        html += `<code style="font-size:12px;">${star} [${r.action.padEnd(6)}] EV: ${r.winRate.toFixed(1)}%</code><br>`;
                    });

                    document.getElementById('ai-result').innerHTML = html;
                } catch (e) {
                    document.getElementById('ai-result').innerHTML = "Simulation error. Proceed manually.";
                } finally {
                    setButtonsState(true);
                }
            }, 10);
        }

        // ==========================================
        // 4. UI & Canvas

        function buildNumpad() {
            const np = document.getElementById('numpad');
            np.innerHTML = '';
            cardsLayout.forEach(c => {
                let btn = document.createElement('button');
                btn.className = `card-btn ${c.type}`;
                btn.innerHTML = `${c.val}<br><small id="count-${c.val}">0</small>`;
                btn.onclick = () => handleCardInput(c.val);
                np.appendChild(btn);
            });
        }

        function updateUI() {
            let counts = {};
            state.roundDeck.forEach(val => {
                counts[val] = (counts[val] || 0) + 1;
            });

            cardsLayout.forEach(c => {
                let val = parseCard(c.val);
                let count = counts[val] || 0;
                if (val === 10) count = Math.floor(count / 4); 
                let el = document.getElementById(`count-${c.val}`);
                if(el && el.innerText !== count.toString()) el.innerText = count;
            });

            document.getElementById('stat-w').innerText = state.stats.wins;
            document.getElementById('stat-l').innerText = state.stats.losses;
            document.getElementById('stat-t').innerText = state.stats.pushes;
            document.getElementById('stat-cards').innerText = state.roundDeck.length; 
            
            document.getElementById('btn-undo').disabled = historyStack.length === 0;

            let processBtn = document.getElementById('btn-process');
            
            if (state.isDropMode) {
                processBtn.disabled = true;
                processBtn.style.background = "#2a2a35";
                processBtn.style.color = "#555";
            } else if (['DEAL_P1', 'DEAL_P2', 'DEAL_D1'].includes(state.phase)) {
                processBtn.innerText = "DEALING...";
                processBtn.disabled = true;
                processBtn.style.background = "#2a2a35";
                processBtn.style.color = "#555";
            } else if (state.phase === 'ROUND_OVER') {
                processBtn.innerText = "START NEW ROUND";
                processBtn.disabled = false;
                processBtn.style.background = "var(--primary)";
                processBtn.style.color = "#000";
            } else {
                processBtn.innerText = "EVALUATE ROUND";
                processBtn.disabled = false;
                processBtn.style.background = "var(--primary)";
                processBtn.style.color = "#000";
            }

            requestAnimationFrame(drawCanvas); 
        }

        function drawCanvas() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            let dScore = getScore(state.dealerHand);
            let pScore = getScore(state.playerHand);
            
            ctx.globalAlpha = state.isDropMode ? 0.2 : 1.0;
            
            ctx.fillStyle = "var(--text-muted)";
            ctx.font = "13px Arial";
            ctx.textAlign = "center";
            ctx.fillText(`DEALER ${dScore > 0 ? `[${dScore}]` : ""}`, canvas.width/2, 25);
            drawHand(state.dealerHand, canvas.width/2, 40);

            ctx.fillStyle = "var(--text-main)";
            ctx.fillText(`PLAYER ${pScore > 0 ? `[${pScore}]` : ""}`, canvas.width/2, 150);
            drawHand(state.playerHand, canvas.width/2, 165);

            ctx.globalAlpha = 1.0; 
        }

        function drawHand(hand, centerX, startY) {
            const cardWidth = 42;
            const cardHeight = 60;
            const spacing = 6;
            const totalWidth = (hand.length * cardWidth) + ((hand.length - 1) * spacing);
            let currentX = centerX - (totalWidth / 2);

            if(hand.length === 0) {
                ctx.strokeStyle = "rgba(255,255,255,0.08)";
                ctx.setLineDash([4, 4]);
                ctx.strokeRect(centerX - cardWidth/2, startY, cardWidth, cardHeight);
                ctx.setLineDash([]);
                return;
            }

            hand.forEach(val => {
                ctx.fillStyle = "#ffffff";
                ctx.beginPath();
                ctx.roundRect(currentX, startY, cardWidth, cardHeight, 4);
                ctx.fill();

                let displayVal = formatCardValue(val);
                ctx.fillStyle = (val === 11) ? "#ff1744" : "#121212"; 
                ctx.font = "bold 18px Arial";
                ctx.textAlign = "center";
                ctx.textBaseline = "middle";
                ctx.fillText(displayVal, currentX + (cardWidth/2), startY + (cardHeight/2));

                currentX += cardWidth + spacing;
            });
        }

        function updateBanner(text, borderColor, textColor) {
            let banner = document.getElementById('banner');
            if(banner.innerText !== text) {
                banner.innerText = text;
                banner.style.borderColor = borderColor;
                banner.style.color = textColor;
                banner.style.background = `rgba(${hexToRgb(borderColor)}, 0.1)`;
            }
        }

        function hexToRgb(hex) {
            let c = hex.substring(1).split('');
            if(c.length === 3){ c = [c[0], c[0], c[1], c[1], c[2], c[2]]; }
            c = '0x' + c.join('');
            return [(c>>16)&255, (c>>8)&255, c&255].join(',');
        }

        function setButtonsState(enabled) {
            let actHit = document.getElementById('act-hit');
            let actStand = document.getElementById('act-stand');
            let actDouble = document.getElementById('act-double');
            let actSplit = document.getElementById('act-split');
            
            let cardBtns = document.querySelectorAll('.card-btn');
            
            if (state.phase === 'ROUND_OVER' && !state.isDropMode) {
                cardBtns.forEach(b => b.disabled = true);
            } else {
                cardBtns.forEach(b => b.disabled = false); 
            }

            if (state.isDropMode) enabled = false; 

            actHit.disabled = !enabled;
            actStand.disabled = !enabled;

            if(enabled) {
                let hand = state.playerHand;
                actDouble.disabled = !(hand.length === 2);
                actSplit.disabled = !(hand.length === 2 && hand[0] === hand[1]);
            } else {
                actDouble.disabled = true;
                actSplit.disabled = true;
            }
        }

        // ==========================================
        // 5. Workflow & Drop Mode Logic
        // ==========================================
        
        function toggleDropMode() {
            let btn = document.getElementById('btn-toggle-drop');
            let overlay = document.getElementById('overlay-discard');
            
            if (!state.isDropMode) {
                saveState(); 
                state.isDropMode = true;
                state.savedPhase = state.phase; 
                state.tempDropped = 0;
                state.droppedThisSession = []; 
                
                btn.innerHTML = "✅ ดำเนินการต่อ (กลับสู่เกม)";
                btn.classList.add('active');
                overlay.style.display = 'block';
                
                updateBanner("DROP MODE: กด Numpad เพื่อเอาไพ่ที่คนอื่นจั่วออกจาก Shoe", "var(--discard-color)", "var(--discard-color)");
                setButtonsState(false);
                updateUI();
                checkPhase(); 
                
            } else {
                state.isDropMode = false;
                state.phase = state.savedPhase; 
                state.savedPhase = null;
                
                btn.innerHTML = "🔥 โหมดทิ้งไพ่คนอื่น (Drop Mode)";
                btn.classList.remove('active');
                overlay.style.display = 'none';
                
                updateUI();
                checkPhase(); 
            }
        }

        function saveState() {
            historyStack.push(JSON.parse(JSON.stringify(state)));
            if (historyStack.length > 50) historyStack.shift();
        }

        function undoAction() {
            if (historyStack.length === 0) return;
            state = historyStack.pop(); 
            
            let btn = document.getElementById('btn-toggle-drop');
            let overlay = document.getElementById('overlay-discard');
            if(state.isDropMode) {
                btn.innerHTML = "✅ ดำเนินการต่อ (กลับสู่เกม)";
                btn.classList.add('active');
                overlay.style.display = 'block';
            } else {
                btn.innerHTML = "🔥 โหมดทิ้งไพ่คนอื่น (Drop Mode)";
                btn.classList.remove('active');
                overlay.style.display = 'none';
            }

            document.getElementById('result-msg').style.display = 'none';
            updateUI();
            checkPhase();
        }

        function fullReset() {
            initDeck();
            startNewRound();
            let btn = document.getElementById('btn-toggle-drop');
            btn.innerHTML = "🔥 โหมดทิ้งไพ่คนอื่น (Drop Mode)";
            btn.classList.remove('active');
            document.getElementById('overlay-discard').style.display = 'none';
            
            updateBanner("Shoe reset successfully.", "#00e676", "#00e676");
            setTimeout(() => checkPhase(), 1500);
        }

        function handleMainBtn() {
            if (state.phase === 'ROUND_OVER') {
                startNewRound();
            } else {
                processRound();
            }
        }

        function handleCardInput(cardStr) {
            if (state.isDropMode) {
                saveState();
                let val = parseCard(cardStr);
                let index = state.roundDeck.indexOf(val);
                if (index > -1) {
                    state.roundDeck.splice(index, 1);
                    state.discardPile.push(val);
                    state.tempDropped++;
                    state.droppedThisSession.push(cardStr);
                }
                updateUI();
                checkPhase(); 
            } else {
                addCard(cardStr);
            }
        }

        function processRound() {
            let uHand = state.playerHand;
            let dHand = state.dealerHand;

            let uScore = getScore(uHand);
            let dScore = getScore(dHand);
            let uBj = (uScore === 21 && uHand.length === 2);
            let dBj = (dScore === 21 && dHand.length === 2);

            let resultText = "";
            let resType = "";

            if (uScore > 21) {
                state.stats.losses++; resultText = "BUST - YOU LOSE"; resType = "loss";
            } else if (dScore > 21) {
                state.stats.wins++; resultText = "DEALER BUSTS - YOU WIN"; resType = "win";
            } else if (uBj && !dBj) {
                state.stats.wins++; resultText = "BLACKJACK - YOU WIN!"; resType = "win";
            } else if (dBj && !uBj) {
                state.stats.losses++; resultText = "DEALER BLACKJACK - YOU LOSE"; resType = "loss";
            } else if (uScore > dScore) {
                state.stats.wins++; resultText = `YOU WIN (${uScore} > ${dScore})`; resType = "win";
            } else if (uScore < dScore) {
                state.stats.losses++; resultText = `YOU LOSE (${uScore} < ${dScore})`; resType = "loss";
            } else {
                state.stats.pushes++; resultText = `PUSH (${uScore} vs ${dScore})`; resType = "push";
            }

            state.phase = 'ROUND_OVER';
            updateUI();
            setButtonsState(false);
            
            const el = document.getElementById('result-msg');
            el.innerText = resultText;
            el.style.display = 'block';
            if(resType === 'win') { el.style.background = 'rgba(0, 230, 118, 0.2)'; el.style.color = 'var(--success)'; }
            else if(resType === 'loss') { el.style.background = 'rgba(255, 23, 68, 0.2)'; el.style.color = 'var(--danger)'; }
            else { el.style.background = 'rgba(255, 234, 0, 0.2)'; el.style.color = 'var(--warning)'; }

            checkPhase();
        }

        function startNewRound() {
            saveState();
            document.getElementById('result-msg').style.display = 'none';

            state.discardPile.push(...state.playerHand);
            state.discardPile.push(...state.dealerHand);
            
            state.playerHand = [];
            state.dealerHand = [];
            state.phase = 'DEAL_P1';
            state.isDropMode = false;
            state.actionTaken = null;
            state.tempDropped = 0;
            state.droppedThisSession = [];

            let btn = document.getElementById('btn-toggle-drop');
            btn.innerHTML = "🔥 โหมดทิ้งไพ่คนอื่น (Drop Mode)";
            btn.classList.remove('active');
            document.getElementById('overlay-discard').style.display = 'none';

            setButtonsState(false);
            document.getElementById('ai-result').innerText = "Waiting for initial deal...";
            updateUI();
            checkPhase();
        }

        function checkPhase() {
            if (state.isDropMode) {
                let dropList = state.droppedThisSession.length > 0 ? state.droppedThisSession.join(', ') : '-';
                document.getElementById('ai-result').innerHTML = `
                    <div style="display:flex; justify-content:space-between; align-items:center;">
                        <b>[โหมดหักไพ่ทิ้ง]</b>
                        <span style="color:var(--text-muted); font-size:12px;">ทิ้งไปแล้ว: ${state.tempDropped} ใบ</span>
                    </div>
                    <div style="margin-top:8px;">ไพ่ที่ทิ้งรอบนี้: <span class="drop-list">${dropList}</span></div>
                    <div style="margin-top:8px; font-size:12px; color:#777;">กด Numpad รัวๆ เพื่อเอาไพ่ออกจากกอง เสร็จแล้วกดดำเนินการต่อ</div>
                `;
                return;
            }

            if (state.phase === 'DEAL_P1') {
                updateBanner("Deal: Select Player's 1st Card", "#00ffcc", "#00ffcc");
            } else if (state.phase === 'DEAL_P2') {
                updateBanner("Deal: Select Player's 2nd Card", "#00ffcc", "#00ffcc");
            } else if (state.phase === 'DEAL_D1') {
                updateBanner("Deal: Select Dealer's Upcard", "#ff007f", "#ff007f");
            } else if (state.phase === 'PLAYING') {
                let score = getScore(state.playerHand);
                if (score >= 21) {
                    state.phase = 'DEALER_TURN';
                    checkPhase();
                } else {
                    updateBanner("Your Turn: Select Action", "#00e676", "#00e676");
                    runAIAnalysis();
                }
            } else if (state.phase === 'WAITING_CARD') {
                updateBanner(`Action [${state.actionTaken}]: Enter drawn card`, "#29b6f6", "#29b6f6");
            } else if (state.phase === 'DEALER_TURN') {
                setButtonsState(false);
                document.getElementById('ai-result').innerText = "Dealer's turn. Input cards they draw, then click 'EVALUATE ROUND'.";
                updateBanner("Dealer's Turn: Enter drawn cards", "#ffea00", "#ffea00");
            } else if (state.phase === 'ROUND_OVER') {
                updateBanner("Round Over: Click 'START NEW ROUND'", "#ffffff", "#ffffff");
                document.getElementById('ai-result').innerText = "Waiting to start next round...";
            }
        }

        function addCard(cardStr) {
            saveState();
            let val = parseCard(cardStr);
            
            let index = state.roundDeck.indexOf(val);
            if (index > -1) state.roundDeck.splice(index, 1);

            if (state.phase === 'DEAL_P1') {
                state.playerHand.push(val);
                state.phase = 'DEAL_P2';
            } else if (state.phase === 'DEAL_P2') {
                state.playerHand.push(val);
                state.phase = 'DEAL_D1';
            } else if (state.phase === 'DEAL_D1') {
                state.dealerHand.push(val);
                state.phase = 'PLAYING';
            } else if (state.phase === 'WAITING_CARD') {
                state.playerHand.push(val);
                
                if (state.actionTaken === 'DOUBLE') {
                    state.phase = 'DEALER_TURN';
                } else {
                    state.phase = 'PLAYING';
                }
            } else if (state.phase === 'DEALER_TURN') {
                state.dealerHand.push(val);
            }

            updateUI();
            checkPhase();
        }

        function recordAction(action) {
            saveState();
            state.actionTaken = action;
            setButtonsState(false);
            
            if (action === 'STAND') {
                state.phase = 'DEALER_TURN';
            } else if (action === 'HIT' || action === 'DOUBLE') {
                state.phase = 'WAITING_CARD';
            } else if (action === 'SPLIT') {
                state.playerHand.pop(); 
                state.phase = 'WAITING_CARD';
            }
            
            updateUI();
            checkPhase();
        }

        document.addEventListener('keydown', function(event) {
            if (event.key.toLowerCase() === 'z' && !document.getElementById('btn-undo').disabled) {
                undoAction();
            } else if (event.key.toLowerCase() === 'd') {
                toggleDropMode();
            }
        });

        buildNumpad();
        initDeck();
        startNewRound();

    </script>
